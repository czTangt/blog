level 11

od -w100 -c /flag | sed ‘s# ##g’

  • 这里的 -w100 指每行显示 100 个字节，-c 以字符形式显示 sed 工具可以进行文本替换，s 表示替换，#是分隔符，##表示要替换的内容为空字符串，g 表示全局替换，即一行中的所有匹配都替换
  • 这里注意管道符的作用，它将前面的输出作为后面的输出，进行更改之后再向 shell 中输出

level 12

hexdump -C /flag | awk -F "|" '{print $2}' | xargs | sed "s# ##g"

  • awk -F "|" '{print $2}'，awk 用于处理上一个语句的输出，使用“|”作为字段分隔符，并打印第二个字段，这里就是按照 | 把每行分成了 3 份。这里就是把两个 | 中的语句提取出来打印
  • xargs 构建并执行来自标准输入的命令行，就是从标准输入中得到数据。这里就是获取 awk 的数据，将其作为参数传递给下一个命令 —> 注意它与管道符的区别
  • sed "s# ##g" 全局替换空格为无，这里首先使用 xargs 把内容输入到 sed 中，之后再进行处理

echo $(xxd -p /flag) | xxd -r -p—> xxd -p /flag | xxd -r -p

level 32

socat 的使用

socat -u "file:/flag" - / socat - /flag

  • -u: 表示无缓存模式，在这个模式下，数据将立即传递而无需等待缓冲区填满
  • "file:/flag": 这是 socat 的源地址，指定了要连接的文件路径。在这里，/flag 是文件路径。
  • -: 这是 socat 的目标地址，表示将数据发送到标准输出(stdout)
  • 这个命令的作用是将文件 /flag 的内容通过无缓冲模式输出到标准输出

whiptail 是一个在 shell 脚本中创建对话框的工具，它通常用于创建用户友好的文本界面

whiptail --textbox "/flag" 20 60

  • whiptail 用于在 shell 脚本中创建对话框
  • --textbox "/flag": 指定要在对话框中显示的文件路径为 “/flag”。--textbox 选项告诉 whiptail 显示文件内容
  • 20 60: 这两个参数分别指定了对话框的高度和宽度。在这里，高度为 20 行，宽度为 60 列

level 51

这里利用的就是 suid 提权之 so 文件注入

基础知识

原理：某些程序运行时候，有着管理员权限，却可以被任何用户调用，这就是 suid 权限。如果这个 suid 程序调用了某些 shared object，可以通过伪造 shared object，来伪造命令。以该程序的 root 权限，去执行提权命令。 cp /bin/bash /tmp/bash && chmod +s /tmp/bash && /tmp/bash -p。注意，suid 权限必须是二进制程序才能用，脚本程序是无效的。

ssh-keygen 可以使用 -D 参数直接运行任意的共享库。如果有 suid 的话就能运行我们的恶意代码造成提权，共享库的创建方式见下方。

1

ssh-keygen -D ./file.s0

gcc 可以通过 -shared 参数来创建共享库，共享库不能单独运行，它相当于一个必须被别人调用才能运行的程序，它与普通二进制程序的区别可以通过 file 命令来查看，共享库没有 interpreter 这样的字段，也就是没有链接解释器。

1

gcc file.c -shared -o file.so

ldd 用于打印程序或者库文件所依赖的共享库列表。我们可以利用它来查看是否有可操作的 os 共享库，注入我们的代码。另一种方法就是利用 strace 来查看程序运行全过程，然后使用 grep 来过滤。

1
2

ldd /usr/bin/ssh-keygen
strace /usr/bin/ssh-keygen 2>&1 | grep -i -E "open|access|no such file"

#include <unistd.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <stdlib.h>
#include <stdio.h>
#include <fcntl.h>
#include <sys/wait.h>

void pwncollege()
{
    return;
}

int main()
{
    int fd[2];
    pipe(fd);//注意管道的建立必须在父子进程建立之前，否则它们不会共用管道
    int pid = fork();
    
    if (pid == 0)//子进程
    {
        close(fd[0]);//关闭管道的读，使管道只能被写入，便于下面写入管道
        dup2(fd[1], STDOUT_FILENO);//dup2将标准输出(STDOUT_FILENO)重定向到管道的写入端 fd[1]。这样下面的输出将不再是默认终端输出，而是写入管道中
        execl("/challenge/embryoio_level60", "embryoio_level60", NULL);
    }
    else//父进程
    {
        int fpid = fork();
        if (fpid == 0)
        {
            close(fd[1]);//关闭管道写，同上
            dup2(fd[0], STDIN_FILENO);//dup2将标准输入(STDIN_FILENO)重定向到管道的读取端 fd[0]，则下面cat的输入就不是来自默认终端输入，而是从管道中读取
            execl("/usr/bin/cat", "cat", NULL);
        }
        else{
            wait(NULL);
        }
    }
    return 0;
}

​ 通过这样的重定向，子进程 1 的输出会成为子进程 2 的输入，实现了进程之间的管道通信。在这个具体的例子中，子进程 1 执行的程序是 /challenge/embryoio_level60，它的输出通过管道传递给子进程 2，而子进程 2 执行的程序是 /usr/bin/cat，将管道的内容输出到标准输出。

#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>
#include <string.h>
#include <sys/socket.h>
#include <arpa/inet.h>
#define SERVER_IP "0.0.0.0"
#define SERVER_PORT 1564
#define BUFFER_SIZE 1024
int main() {
    int sockfd;
    struct sockaddr_in server_addr;
    char buffer[BUFFER_SIZE];
    char ans[BUFFER_SIZE];
    // 创建套接字
    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd == -1) {
        perror("socket");
        exit(EXIT_FAILURE);
    }

    // 设置服务器地址
    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = inet_addr(SERVER_IP);
    server_addr.sin_port = htons(SERVER_PORT);

    // 连接到服务器
    if (connect(sockfd, (struct sockaddr *)&server_addr, sizeof(server_addr)) == -1) {
        perror("connect");
        exit(EXIT_FAILURE);
    }

    while(1)
    {
        memset(buffer, 0, BUFFER_SIZE);
        if (recv(sockfd, buffer, BUFFER_SIZE, 0) == -1) {
            perror("recv");
            exit(EXIT_FAILURE);
        }
        if(strlen(buffer)>0)
        {
            printf("Received: %s\n", buffer);
            if(!memcmp("[TEST] CHALLENGE! Please send the solution",buffer,strlen(("[TEST] CHALLENGE! Please send the solution"))))
            {
                
                memset(ans, 0, BUFFER_SIZE);
                read(0,ans,BUFFER_SIZE);
                if (send(sockfd, ans, strlen(ans), 0) == -1) {
                    perror("send");
                    exit(EXIT_FAILURE);
                }
            }

        }
    }
    close(sockfd);
    return 0;
}
